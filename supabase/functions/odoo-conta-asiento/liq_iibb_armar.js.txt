const num = (s) => Number(s.replace(/\./g, "").replace(",", "."));
const r2 = (x) => Math.round(x * 100) / 100;
const norm = (s) => s.normalize("NFD").replace(/[\u0300-\u036f]/g, "").toUpperCase().trim();
function leer(f) {
  const t = execFileSync("pdftotext", ["-raw", D + f, "-"]).toString("latin1");
  const sec = /Secuencia: ([^\n]+)/.exec(t)[1].trim(), tx = (/Transacci.n: (\d+)/.exec(t) || [])[1];
  const bloque = t.slice(t.indexOf("A Pagar"), t.indexOf("TOTALES")).replace(/SANTIAGO DEL\s*\n\s*ESTERO\s*\n/, "SANTIAGO DEL ESTERO ");
  const filas = {};
  for (const m of bloque.matchAll(/^([A-Z ]+?) \$ ([\d.,]+) \$ ([\d.,]+) \$ ([\d.,]+) \$ ([\d.,]+) \$ ([\d.,]+) \$ ([\d.,]+)$/gm))
    filas[norm(m[1])] = { imp: num(m[2]), restan: num(m[3]), suman: num(m[4]), af: num(m[5]), aff: num(m[6]), pagar: num(m[7]) };
  const tot = /TOTALES \$ ([\d.,]+) \$ ([\d.,]+) \$ ([\d.,]+) \$ ([\d.,]+) \$ ([\d.,]+) \$ ([\d.,]+)/.exec(t);
  return { sec, tx, filas, totPagar: num(tot[6]), totImp: num(tot[1]) };
}
